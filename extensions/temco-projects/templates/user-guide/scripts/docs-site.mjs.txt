// Renders the site a reader gets: `dist/docs`, one HTML page per shipped topic.
//
// MSDN shape on purpose — a breadcrumb, the whole table of contents at the left with the current branch open, an
// on-page list of sections, "Applies to", previous/next in reading order, and a permalink on every heading. The
// metadata is the registry's: `order` is the TOC, `ship` decides what is published, `title` is the heading and the
// breadcrumb, `audience` is the "Applies to" line.
//
// Temco Platform's Build Center runs this as the `docs:site` script and judges it by the files it leaves behind.

import { mkdirSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

const root = process.cwd();
const registry = readJson(join(root, 'docs', 'help', 'registry.json')) ?? { areas: {}, pages: {} };
const pkg = readJson(join(root, 'package.json')) ?? {};
const version = String(pkg.version ?? '');
const out = join(root, 'dist', 'docs');

const pages = Object.entries(registry.pages ?? {})
	.map(([id, page]) => ({ id, ...page }))
	.filter((page) => page.ship !== false)
	.sort((left, right) =>
		String(left.area).localeCompare(String(right.area))
		|| String(left.chapter).localeCompare(String(right.chapter))
		|| (left.order ?? 0) - (right.order ?? 0)
		|| left.id.localeCompare(right.id));

rmSync(out, { recursive: true, force: true });
mkdirSync(out, { recursive: true });

for (let index = 0; index < pages.length; index++) {
	const page = pages[index];
	const rendered = render(read(join(root, 'docs', page.id + '.md')));
	const title = page.title ?? page.id;
	const html = shell({
		title,
		breadcrumb: [areaTitle(page.area), page.chapter && page.chapter !== '(root)' ? titleCase(page.chapter) : '', title].filter(Boolean),
		toc: toc(pages, page.id, registry.areas ?? {}),
		sectionList: rendered.sections,
		body: rendered.html,
		appliesTo: page.audience ?? version,
		previous: index > 0 ? pages[index - 1] : undefined,
		next: index < pages.length - 1 ? pages[index + 1] : undefined
	});
	const file = join(out, page.id + '.html');
	mkdirSync(join(file, '..'), { recursive: true });
	writeFileSync(file, html);
}

writeFileSync(join(out, 'index.html'), landing(pages, registry.areas ?? {}));
console.log('site: ' + pages.length + ' page(s) in ' + join('dist', 'docs'));

function read(file) {
	try {
		return readFileSync(file, 'utf8');
	} catch {
		return '';
	}
}

function readJson(file) {
	try {
		return JSON.parse(readFileSync(file, 'utf8'));
	} catch {
		return undefined;
	}
}

function areaTitle(area) {
	return titleCase(area);
}

function titleCase(text) {
	const words = String(text ?? '').replace(/[-_/]+/g, ' ').replace(/\s+/g, ' ').trim();
	return words.length === 0 ? '' : words.replace(/\b[a-z]/g, (letter) => letter.toUpperCase());
}

function slug(text) {
	return String(text).toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '') || 'section';
}

function escapeHtml(text) {
	return String(text)
		.replace(/&/g, '&amp;')
		.replace(/</g, '&lt;')
		.replace(/>/g, '&gt;')
		.replace(/"/g, '&quot;');
}

/** The Markdown these pages are written in: headings, paragraphs, lists, fences, inline code, links, emphasis. */
function render(markdown) {
	const lines = String(markdown).replace(/\r\n/g, '\n').split('\n');
	const html = [];
	const sections = [];
	let paragraph = [];
	let list = false;
	let fence = false;
	let code = [];

	const flush = () => {
		if (paragraph.length > 0) {
			html.push('<p>' + inline(paragraph.join(' ')) + '</p>');
			paragraph = [];
		}
		if (list) {
			html.push('</ul>');
			list = false;
		}
	};

	for (const line of lines) {
		if (/^```/.test(line)) {
			if (fence) {
				html.push('<pre><code>' + escapeHtml(code.join('\n')) + '</code></pre>');
				code = [];
				fence = false;
			} else {
				flush();
				fence = true;
			}
			continue;
		}
		if (fence) {
			code.push(line);
			continue;
		}
		if (/^#{1,6}\s/.test(line)) {
			flush();
			const level = /^#+/.exec(line)[0].length;
			const text = line.replace(/^#+\s*/, '');
			const id = slug(text);
			if (level === 2 || level === 3) {
				sections.push({ id, text });
			}
			html.push('<h' + level + ' id="' + id + '">' + inline(text) + ' <a class="anchor" href="#' + id + '" aria-label="Permalink">#</a></h' + level + '>');
			continue;
		}
		if (/^\s*[-*]\s+/.test(line)) {
			if (paragraph.length > 0) {
				flush();
			}
			if (!list) {
				html.push('<ul>');
				list = true;
			}
			html.push('<li>' + inline(line.replace(/^\s*[-*]\s+/, '')) + '</li>');
			continue;
		}
		if (line.trim().length === 0) {
			flush();
			continue;
		}
		paragraph.push(line.trim());
	}
	flush();
	return { html: html.join('\n'), sections };
}

function inline(text) {
	let out = escapeHtml(text);
	out = out.replace(/`([^`]+)`/g, '<code>$1</code>');
	out = out.replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
	out = out.replace(/\[([^\]]+)\]\(([^)]+)\)/g, (match, label, href) =>
		'<a href="' + String(href).replace(/\.md$/i, '.html') + '">' + label + '</a>');
	return out;
}

/** The table of contents: area → chapter → topic, the current topic marked and its branch open. */
function toc(pages, currentId, areas) {
	const groups = new Map();
	for (const page of pages) {
		const area = page.area ?? 'overview';
		if (!groups.has(area)) {
			groups.set(area, new Map());
		}
		const chapters = groups.get(area);
		const chapter = page.chapter && page.chapter !== '(root)' ? page.chapter : '';
		if (!chapters.has(chapter)) {
			chapters.set(chapter, []);
		}
		chapters.get(chapter).push(page);
	}

	const out = [];
	for (const [area, chapters] of groups) {
		out.push('<div class="toc__area"><span>' + escapeHtml(areas[area]?.title ?? areaTitle(area)) + '</span></div>');
		for (const [chapter, topics] of chapters) {
			if (chapter) {
				out.push('<div class="toc__chapter">' + escapeHtml(titleCase(chapter)) + '</div>');
			}
			for (const topic of topics) {
				const current = topic.id === currentId ? ' class="current" aria-current="page"' : '';
				out.push('<a href="' + topic.id + '.html"' + current + '>' + escapeHtml(topic.title ?? topic.id) + '</a>');
			}
		}
	}
	return out.join('\n');
}

function shell(options) {
	const sections = options.sectionList.length === 0
		? ''
		: '<nav class="onpage" aria-label="On this page"><div class="onpage__title">On this page</div>'
			+ options.sectionList.map((section) => '<a href="#' + section.id + '">' + escapeHtml(section.text) + '</a>').join('')
			+ '</nav>';

	const pager = '<nav class="pager">'
		+ (options.previous ? '<a href="' + options.previous.id + '.html" rel="prev">← ' + escapeHtml(options.previous.title ?? options.previous.id) + '</a>' : '<span></span>')
		+ (options.next ? '<a href="' + options.next.id + '.html" rel="next">' + escapeHtml(options.next.title ?? options.next.id) + ' →</a>' : '<span></span>')
		+ '</nav>';

	return '<!doctype html><html lang="en"><head><meta charset="utf-8">'
		+ '<meta name="viewport" content="width=device-width,initial-scale=1">'
		+ '<title>' + escapeHtml(options.title) + '</title>'
		+ '<link rel="stylesheet" href="site.css">'
		+ '</head><body>'
		+ '<header class="top">' + escapeHtml(pkg.name ?? 'Documentation') + '</header>'
		+ '<div class="layout">'
		+ '<nav class="toc" aria-label="Contents">' + options.toc + '</nav>'
		+ '<main><nav class="crumbs" aria-label="Breadcrumb">' + options.breadcrumb.map(escapeHtml).join(' / ') + '</nav>'
		+ '<h1>' + escapeHtml(options.title) + '</h1>'
		+ (options.appliesTo ? '<div class="applies">Applies to: ' + escapeHtml(options.appliesTo) + '</div>' : '')
		+ options.body
		+ pager
		+ '<footer class="feedback">Was this page helpful? <a href="mailto:docs@example.invalid?subject=' + encodeURIComponent(options.title) + '">Tell us</a></footer>'
		+ '</main>'
		+ (sections ? '<aside class="rail">' + sections + '</aside>' : '')
		+ '</div>'
		+ '<link rel="stylesheet" href="site.css">'
		+ '</body></html>';
}

function landing(pages, areas) {
	const links = pages.map((page) =>
		'<li><a href="' + page.id + '.html">' + escapeHtml(page.title ?? page.id) + '</a> <span>' + escapeHtml(areaTitle(page.area)) + '</span></li>').join('');
	return '<!doctype html><html lang="en"><head><meta charset="utf-8"><title>' + escapeHtml(pkg.name ?? 'Documentation')
		+ '</title><link rel="stylesheet" href="site.css"></head><body><header class="top">'
		+ escapeHtml(pkg.name ?? 'Documentation') + '</header><main class="landing"><h1>' + escapeHtml(pkg.name ?? 'Documentation')
		+ '</h1><p>Everything this project publishes, in reading order.</p><ul>' + links + '</ul></main></body></html>';
}
